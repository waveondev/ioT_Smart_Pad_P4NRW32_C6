cmake_minimum_required(VERSION 3.16)

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

# 모든 하위 컴포넌트(espressif__esp_hosted 포함)에 CMake 리스트 형태로 전역 매크로 주입
idf_build_set_property(COMPILE_DEFINITIONS 
    "CONFIG_ESP_WIFI_STATIC_RX_BUFFER_NUM=10;"
    "CONFIG_ESP_WIFI_DYNAMIC_RX_BUFFER_NUM=32;"
    "CONFIG_ESP_WIFI_TX_BUFFER_TYPE=1;"
    "CONFIG_ESP_WIFI_DYNAMIC_RX_MGMT_BUF=1;"
    "CONFIG_ESP_WIFI_ESPNOW_MAX_ENCRYPT_NUM=7;"
    "CONFIG_ESP_WIFI_DYNAMIC_TX_BUFFER_NUM=32;"
    "CONFIG_ESP_WIFI_AMPDU_TX_ENABLED=1;"
    "CONFIG_ESP_WIFI_TX_BA_WIN=6;"
    "CONFIG_ESP_WIFI_AMPDU_RX_ENABLED=1;"
    "CONFIG_ESP_WIFI_RX_BA_WIN=6;"
    "CONFIG_ESP_WIFI_SOFTAP_SUPPORT=1;"
    "ESP_WIFI_MAX_CONN_NUM=10"
    APPEND
)

project(ioT_Smart_Pad)

# project() 선언 이후 생성되는 타깃에 대한 전역 C 컴파일 옵션 세팅
add_compile_options(
    -DCONFIG_ESP_WIFI_STATIC_RX_BUFFER_NUM=10
    -DCONFIG_ESP_WIFI_DYNAMIC_RX_BUFFER_NUM=32
    -DCONFIG_ESP_WIFI_TX_BUFFER_TYPE=1
    -DCONFIG_ESP_WIFI_DYNAMIC_RX_MGMT_BUF=1
    -DCONFIG_ESP_WIFI_ESPNOW_MAX_ENCRYPT_NUM=7
    -DCONFIG_ESP_WIFI_DYNAMIC_TX_BUFFER_NUM=32
    -DCONFIG_ESP_WIFI_AMPDU_TX_ENABLED=1
    -DCONFIG_ESP_WIFI_TX_BA_WIN=6
    -DCONFIG_ESP_WIFI_AMPDU_RX_ENABLED=1
    -DCONFIG_ESP_WIFI_RX_BA_WIN=6
    -DCONFIG_ESP_WIFI_SOFTAP_SUPPORT=1
    -DESP_WIFI_MAX_CONN_NUM=10
)
